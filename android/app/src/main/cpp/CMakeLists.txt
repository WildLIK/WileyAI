cmake_minimum_required(VERSION 3.31.6)
project(wiley_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../../local/android-native/llama.cpp-8642541b255bfb5dc39cd351a178be09013a55d6")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run scripts/prepare-android-native.py before building")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama-build)
add_library(wiley_llm SHARED wiley_llm.cpp)
target_link_libraries(wiley_llm PRIVATE llama android log)
target_link_options(wiley_llm PRIVATE "-Wl,-z,max-page-size=16384")
